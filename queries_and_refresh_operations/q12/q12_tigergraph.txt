CREATE OR REPLACE OPENCYPHER QUERY q12_tigergraph(
    STRING mode1,
    STRING mode2,
    DATETIME start_date,
    DATETIME end_date
) FOR GRAPH TPCH_G {

  MATCH
    (l:G_LINEITEM)
      -[:G_LINEITEM_ORDERS]->(o:G_ORDERS)

  WHERE l.l_shipmode IN [$mode1, $mode2]
    AND l.l_commitdate < l.l_receiptdate
    AND l.l_shipdate < l.l_commitdate
    AND l.l_receiptdate >= $start_date
    AND l.l_receiptdate <  $end_date

  RETURN
    l.l_shipmode AS l_shipmode,

    SUM(
      CASE
        WHEN o.o_orderpriority = "1-URGENT"
          OR o.o_orderpriority = "2-HIGH"
        THEN 1
        ELSE 0
      END
    ) AS high_line_count,

    SUM(
      CASE
        WHEN o.o_orderpriority <> "1-URGENT"
          AND o.o_orderpriority <> "2-HIGH"
        THEN 1
        ELSE 0
      END
    ) AS low_line_count

  ORDER BY l_shipmode
}
