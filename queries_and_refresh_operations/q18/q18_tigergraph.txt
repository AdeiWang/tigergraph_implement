CREATE OR REPLACE OPENCYPHER QUERY q18_tigergraph(
    FLOAT quantity_threshold
) FOR GRAPH TPCH_G {

  MATCH
    (l:G_LINEITEM)
      -[:G_LINEITEM_ORDERS]->(o:G_ORDERS)
      -[:G_ORDERS_CUSTOMER]->(c:G_CUSTOMER)

  WITH
    c.c_name AS customer_name,
    c.c_custkey AS customer_key,
    o.o_orderkey AS order_key,
    o.o_orderdate AS order_date,
    o.o_totalprice AS price,
    SUM(l.l_quantity) AS amount

  WHERE amount > $quantity_threshold

  RETURN
    customer_name,
    customer_key,
    order_key,
    order_date,
    price,
    amount

  ORDER BY
    price DESC,
    order_date ASC
}
